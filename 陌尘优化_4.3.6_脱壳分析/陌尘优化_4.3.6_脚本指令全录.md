shizuku_exec() {
    cmd="$1"
    local apk_path=$(pm path moe.shizuku.privileged.api 2>/dev/null | grep package: | cut -d: -f2 | cut -d" " -f1)
    if [ -z "$apk_path" ]; then
        apk_path="/data/app/moe.shizuku.privileged.api-*/base.apk"
    fi
    app_process -Djava.class.path="$apk_path" /system/bin --nice-name=shizuku_daemon \
        moe.shizuku.privileged.api.Server "$cmd" >/dev/null 2>&1
}
