===== app/src/main/java/cn/sky/angin/client/crypto/CryptoInterceptor.kt =====
package cn.sky.angin.client.crypto

import android.util.Base64
import okhttp3.Interceptor
import okhttp3.MediaType.Companion.toMediaType
import okhttp3.OkHttpClient
import okhttp3.Request
import okhttp3.RequestBody
import okhttp3.RequestBody.Companion.toRequestBody
import okhttp3.Response
import okio.Buffer
import java.security.SecureRandom
import javax.crypto.Cipher
import javax.crypto.spec.GCMParameterSpec
import javax.crypto.spec.SecretKeySpec

class CryptoInterceptor(private val client: OkHttpClient) : Interceptor {
    private val secureRandom = SecureRandom()
    private var cachedKey: ByteArray? = null

    @Synchronized
    private fun getKey(): ByteArray? {
        cachedKey?.let { return it }
        val keyRequest = Request.Builder()
            .url("https://sky.angin.cn/api/crypto/key")
            .get()
            .build()
        client.newCall(keyRequest).execute().use { response ->
            if (!response.isSuccessful) return null
            val body = response.body?.string() ?: return null
            // Expected: {"success":true,"key":"base64"}
            val keyRegex = """"key"\s*:\s*"([^"]+)"""".toRegex()
            val match = keyRegex.find(body) ?: return null
            val keyB64 = match.groupValues[1]
            cachedKey = Base64.decode(keyB64, Base64.DEFAULT)
        }
        return cachedKey
    }

    private fun encrypt(plainText: String): String? {
        val key = getKey() ?: return null
        val cipher = Cipher.getInstance("AES/GCM/NoPadding")
        val iv = ByteArray(12).also { secureRandom.nextBytes(it) }
        cipher.init(Cipher.ENCRYPT_MODE, SecretKeySpec(key, "AES"), GCMParameterSpec(128, iv))
        val cipherText = cipher.doFinal(plainText.toByteArray(Charsets.UTF_8))
        val combined = ByteArray(iv.size + cipherText.size)
        System.arraycopy(iv, 0, combined, 0, iv.size)
        System.arraycopy(cipherText, 0, combined, iv.size, cipherText.size)
        return Base64.encodeToString(combined, Base64.NO_WRAP)
    }

    private fun decrypt(cipherTextB64: String): String? {
        val key = getKey() ?: return null
        val combined = Base64.decode(cipherTextB64, Base64.DEFAULT)
        if (combined.size < 13) return null
        val iv = combined.copyOfRange(0, 12)
        val cipherText = combined.copyOfRange(12, combined.size)
        val cipher = Cipher.getInstance("AES/GCM/NoPadding")
        cipher.init(Cipher.DECRYPT_MODE, SecretKeySpec(key, "AES"), GCMParameterSpec(128, iv))
        val plainText = cipher.doFinal(cipherText)
        return String(plainText, Charsets.UTF_8)
    }

    override fun intercept(chain: Interceptor.Chain): Response {
        val originalRequest = chain.request()
        var request = originalRequest

        val isApiPath = request.url.encodedPath.contains("/api/")
        val isGetMethod = request.method == "GET"
        val hasBody = request.body != null

        if (isApiPath && !isGetMethod && hasBody) {
            val originalBody = request.body ?: return chain.proceed(request)
            val buffer = Buffer()
            originalBody.writeTo(buffer)
            val plainText = buffer.readUtf8()

            val encryptedBody = encrypt(plainText) ?: return chain.proceed(request)
            val newBody = encryptedBody.toRequestBody("text/plain".toMediaType())

            request = request.newBuilder()
                .header("X-Encrypted", "1")
                .method(request.method, newBody)
                .build()
        }

        val response = chain.proceed(request)

        if (response.headers("X-Encrypted").contains("1")) {
            val encryptedResponseBody = response.body?.string() ?: return response
            val decryptedBody = decrypt(encryptedResponseBody) ?: return response
            val newResponseBody = decryptedBody.toRequestBody(response.body?.contentType())
            return response.newBuilder()
                .body(newResponseBody)
                .build()
        }

        return response
    }
}

package cn.sky.angin.client.net

import android.content.Context
import kotlinx.serialization.json.Json
import okhttp3.MediaType.Companion.toMediaType
import okhttp3.OkHttpClient
import okhttp3.logging.HttpLoggingInterceptor
import retrofit2.Retrofit
import retrofit2.converter.kotlinx.serialization.asConverterFactory
import java.util.concurrent.TimeUnit

object NetworkModule {
    private const val BASE_URL = "https://sky.angin.cn/"

    lateinit var okHttpClient: OkHttpClient
        private set

    lateinit var retrofit: Retrofit
        private set

    lateinit var apiService: ApiService
        private set

    val json = Json {
        ignoreUnknownKeys = true
        coerceInputValues = true
        encodeDefaults = true
    }

    fun init(context: Context) {
        val logging = HttpLoggingInterceptor().apply {
            level = HttpLoggingInterceptor.Level.BODY
        }

        val baseClient = OkHttpClient.Builder()
            .connectTimeout(30, TimeUnit.SECONDS)
            .readTimeout(30, TimeUnit.SECONDS)
            .writeTimeout(30, TimeUnit.SECONDS)
            .addInterceptor(logging)
            .cookieJar(CookieJarProvider(context))
            .build()

        val cryptoInterceptor = CryptoInterceptor(baseClient)

        okHttpClient = baseClient.newBuilder()
            .addInterceptor(cryptoInterceptor)
            .build()

        retrofit = Retrofit.Builder()
            .baseUrl(BASE_URL)
            .client(okHttpClient)
            .addConverterFactory(json.asConverterFactory("application/json".toMediaType()))
            .build()

        apiService = retrofit.create(ApiService::class.java)
    }
}

===== app/src/main/java/cn/sky/angin/client/net/CookieJarProvider.kt =====
package cn.sky.angin.client.net

import android.content.Context
import okhttp3.Cookie
import okhttp3.CookieJar
import okhttp3.HttpUrl
import java.util.concurrent.ConcurrentHashMap

class CookieJarProvider(private val context: Context) : CookieJar {
    private val cookieStore = ConcurrentHashMap<String, List<Cookie>>()

    override fun saveFromResponse(url: HttpUrl, cookies: List<Cookie>) {
        cookieStore[url.host] = cookies
    }

    override fun loadForRequest(url: HttpUrl): List<Cookie> {
        return cookieStore[url.host] ?: emptyList()
    }
}

package cn.sky.angin.client.net

import kotlinx.serialization.Serializable
import retrofit2.http.Body
import retrofit2.http.GET
import retrofit2.http.POST
import retrofit2.http.Path

@Serializable
data class LoginRequest(
    val account: String,
    val login_type: String,
    val password: String,
    val device_id: String
)

@Serializable
data class BaseResponse<T>(
    val success: Boolean = false,
    val message: String? = null,
    val data: T? = null
)

interface ApiService {
    @GET("api/config/site")
    suspend fun getSiteConfig(): BaseResponse<Map<String, String>>

    @GET("api/auth/me")
    suspend fun getAuthMe(): BaseResponse<String>

    @POST("api/auth/login")
    suspend fun login(@Body req: LoginRequest): BaseResponse<String>

    @POST("api/auth/logout")
    suspend fun logout(): BaseResponse<String>

    @POST("api/accounts/add/start")
    suspend fun addAccountStart(@Body req: LoginRequest): BaseResponse<String>

    @GET("api/accounts/add/status/{task_id}")
    suspend fun getAddAccountStatus(@Path("task_id") taskId: String): BaseResponse<String>
}

package cn.sky.angin.client.ui.theme

import androidx.compose.material3.lightColorScheme
import androidx.compose.runtime.Composable

private val DarkColorScheme = darkColorScheme(
    primary = Color(0xFF82B1FF),
    secondary = Color(0xFF448AFF),
    tertiary = Color(0xFF2979FF)
)

private val LightColorScheme = lightColorScheme(
    primary = Color(0xFF448AFF),
    secondary = Color(0xFF82B1FF),

    darkTheme: Boolean = true,
    content: @Composable () -> Unit
