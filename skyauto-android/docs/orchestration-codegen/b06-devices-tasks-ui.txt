package com.skyauto.app.data

data class Device(
    val id: String,
    val name: String,
    val model: String,
    val status: String,
    val online: Boolean,
    val accountIds: List<String>
)

data class Account(
    val id: String,
    val displayName: String,
    val account: String,
    val platform: String
)

data class TaskType(
    val id: String,
    val name: String,
    val reward: String,
    val needsManualConfirm: Boolean,
    val category: String
)

data class Task(
    val id: String,
    val type: String,
    val typeName: String,
    val status: String,
    val progress: Int,
    val reward: String,
    val message: String,
    val accountIds: List<String>,
    val createdAt: String
) {
    val isTerminal: Boolean
        get() = status in setOf("success", "completed", "failed", "claimed", "canceled", "expired")

    val canClaim: Boolean
        get() = status in setOf("success", "completed")

    val needsManualConfirm: Boolean
        get() {
            val manual = setOf("sms", "bilibili", "huawei")
            val typeHit = manual.any { type.contains(it, ignoreCase = true) || typeName.contains(it, ignoreCase = true) }
            val statusHit = status.startsWith("need_") ||
                status.contains("confirm", ignoreCase = true) ||
                status.contains("verify", ignoreCase = true) ||
                status.contains("manual", ignoreCase = true)
            return typeHit && statusHit
        }
}

enum class PollState {
    IDLE,
    POLLING,
    WAITING_MANUAL,
    DONE,
    ERROR
}

package com.skyauto.app.data

import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.withContext
import okhttp3.JavaNetCookieJar
import okhttp3.MediaType.Companion.toMediaType
import okhttp3.OkHttpClient
import okhttp3.Request
import okhttp3.RequestBody.Companion.toRequestBody
import org.json.JSONObject
import java.net.CookieManager
import java.security.SecureRandom
import java.util.Base64
import java.util.concurrent.TimeUnit
import javax.crypto.Cipher
import javax.crypto.spec.GCMParameterSpec
import javax.crypto.spec.SecretKeySpec
import javax.crypto.SecretKey

object ApiClient {

    private const val BASE_URL = "https://sky.angin.cn"

    private val cookieManager = CookieManager()

    private val client = OkHttpClient.Builder()
        .cookieJar(JavaNetCookieJar(cookieManager))
        .connectTimeout(15, TimeUnit.SECONDS)
        .readTimeout(15, TimeUnit.SECONDS)
        .writeTimeout(15, TimeUnit.SECONDS)
        .build()

    @Volatile
    var apiRoot: String = "/api"
        private set

    private var aesKey: SecretKey? = null
    private var configCache: Pair<Long, String>? = null

    fun updateApiRoot(root: String) {
        if (root.isNotBlank()) apiRoot = root
    }

    fun clearAuthCache() {
        aesKey = null
        configCache = null
    }

    private suspend fun ensureKey(): SecretKey? = withContext(Dispatchers.IO) {
        aesKey?.let { return@withContext it }

        val url = BASE_URL + apiRoot + "/crypto/key"
        val request = Request.Builder()
            .url(url)
            .header("Accept", "application/json")
            .build()

        client.newCall(request).execute().use { response ->
            if (!response.isSuccessful) return@withContext null
            val json = JSONObject(response.body?.string() ?: "{}")
            if (!json.optBoolean("success") || !json.has("key")) return@withContext null
            val keyBytes = Base64.getDecoder().decode(json.getString("key"))
            SecretKeySpec(keyBytes, "AES")
        }.also { aesKey = it }
    }

    private fun encrypt(plain: String, key: SecretKey): String {
        val iv = ByteArray(12).also { SecureRandom().nextBytes(it) }
        val cipher = Cipher.getInstance("AES/GCM/NoPadding")
        cipher.init(Cipher.ENCRYPT_MODE, key, GCMParameterSpec(128, iv))
        val cipherText = cipher.doFinal(plain.toByteArray(Charsets.UTF_8))
        val out = ByteArray(12 + cipherText.size)
        System.arraycopy(iv, 0, out, 0, 12)
        System.arraycopy(cipherText, 0, out, 12, cipherText.size)
        return Base64.getEncoder().encodeToString(out)
    }

    private fun decrypt(data: String, key: SecretKey): String {
        val bytes = Base64.getDecoder().decode(data)
        val iv = bytes.copyOfRange(0, 12)
        val cipherText = bytes.copyOfRange(12, bytes.size)
        val cipher = Cipher.getInstance("AES/GCM/NoPadding")
        cipher.init(Cipher.DECRYPT_MODE, key, GCMParameterSpec(128, iv))
        return String(cipher.doFinal(cipherText), Charsets.UTF_8)
    }

    suspend fun request(
        path: String,
        method: String = "GET",
        body: String? = null,
        headers: Map<String, String> = emptyMap()
    ): String = withContext(Dispatchers.IO) {
        val fullPath = if (path.startsWith("/")) path else "/$path"
        val url = BASE_URL + fullPath

        val isConfigSite = method.equals("GET", true) && body == null && fullPath.contains("/config/site")
        if (isConfigSite) {
            configCache?.let {
                if (System.currentTimeMillis() - it.first < 10_000) {
                    return@withContext it.second
                }
            }
        }

        val isAuthRequest = !method.equals("GET", true) &&
            listOf("/auth/login", "/auth/logout", "/auth/register").any { fullPath.contains(it) }
        if (isAuthRequest) clearAuthCache()

        val requestBuilder = Request.Builder().url(url)
        headers.forEach { (k, v) -> requestBuilder.header(k, v) }

        var finalBody = body
        var encrypted = false

        if (body != null && !fullPath.contains("/api/crypto/key")) {
            val key = ensureKey() ?: throw Exception("无法获取加密密钥")
            finalBody = encrypt(body, key)
            encrypted = true
            requestBuilder.header("X-Encrypted", "1")
        }

        if (finalBody != null) {
            val mediaType = headers["Content-Type"]?.toMediaType()
            requestBuilder.method(method.uppercase(), finalBody.toRequestBody(mediaType))
        } else {
            requestBuilder.method(method.uppercase(), null)
        }

        if (isAuthRequest) clearAuthCache()

        client.newCall(requestBuilder.build()).execute().use { response ->
            val responseBody = response.body?.string() ?: ""
            val result = if (response.header("X-Encrypted") == "1") {
                val key = ensureKey() ?: throw Exception("无法获取加密密钥")
                decrypt(responseBody, key)
            } else {
                responseBody
            }

            if (!response.isSuccessful) {
                throw Exception("HTTP ${response.code}: ${result.take(200)}")
            }

            if (isConfigSite) {
                configCache = System.currentTimeMillis() to result
            }

            result
        }
    }
}

package com.skyauto.app.data

import org.json.JSONArray
import org.json.JSONObject

class Repository {

    suspend fun fetchSiteConfig() {
        try {
            val body = ApiClient.request("/config/site")
            val root = JSONObject(body)
            val config = root.optJSONObject("config") ?: return
            val endpoints = config.optJSONObject("serverEndpoints")
                ?: config.optJSONObject("server_endpoints")
                ?: return
            val api = endpoints.optString("api")
            if (api.isNotBlank()) ApiClient.updateApiRoot(api)
        } catch (_: Exception) {
        }
    }

    suspend fun getDevices(): List<Device> {
        val body = ApiClient.request("/devices")
        val json = JSONObject(body)
        val array = json.optJSONArray("devices") ?: JSONArray()
        return (0 until array.length()).map { array.getJSONObject(it).toDevice() }
    }

    suspend fun createDevice(name: String, model: String): Device {
        val payload = JSONObject()
            .put("name", name)
            .put("model", model)
            .put("simulate_online", true)
        val body = ApiClient.request(
            "/devices",
            "POST",
            payload.toString(),
            mapOf("Content-Type" to "application/json")
        )
        val json = JSONObject(body)
        return json.optJSONObject("device")?.toDevice() ?: json.toDevice()
    }

    suspend fun deleteDevice(deviceId: String) {
        ApiClient.request("/devices/$deviceId", "DELETE")
    }

    suspend fun setDeviceOnline(deviceId: String, online: Boolean) {
        val payload = JSONObject().put("online", online)
        ApiClient.request(
            "/devices/$deviceId/simulate",
            "POST",
            payload.toString(),
            mapOf("Content-Type" to "application/json")
        )
    }

    suspend fun bindAccount(deviceId: String, accountId: String) {
        val payload = JSONObject()
            .put("action", "bind")
            .put("account_id", accountId)
        ApiClient.request(
            "/devices/$deviceId/accounts",
            "POST",
            payload.toString(),
            mapOf("Content-Type" to "application/json")
        )
    }

    suspend fun unbindAccount(deviceId: String, accountId: String) {
        val payload = JSONObject()
            .put("action", "unbind")
            .put("account_id", accountId)
        ApiClient.request(
            "/devices/$deviceId/accounts",
            "POST",
            payload.toString(),
            mapOf("Content-Type" to "application/json")
        )
    }

    suspend fun getAccounts(): List<Account> {
        val body = ApiClient.request("/accounts")
        val json = JSONObject(body)
        val array = json.optJSONArray("accounts") ?: JSONArray()
        return (0 until array.length()).map { array.getJSONObject(it).toAccount() }
    }

    suspend fun getTaskTypes(): List<TaskType> {
        val body = ApiClient.request("/tasks/types")
        val json = JSONObject(body)
        val array = json.optJSONArray("task_types")
            ?: json.optJSONArray("types")
            ?: JSONArray()
        return (0 until array.length()).map { array.getJSONObject(it).toTaskType() }
    }

    suspend fun getTasks(): List<Task> {
        val body = ApiClient.request("/tasks")
        val json = JSONObject(body)
        val array = json.optJSONArray("tasks") ?: JSONArray()
        return (0 until array.length()).map { array.getJSONObject(it).toTask() }
    }

    suspend fun createTask(
        type: String,
        accountIds: List<String>,
        deviceId: String?
    ): Task {
        val payload = JSONObject()
            .put("task_type", type)
            .put("account_ids", JSONArray(accountIds))
        if (!deviceId.isNullOrBlank()) payload.put("device_id", deviceId)

        val body = ApiClient.request(
            "/tasks",
            "POST",
            payload.toString(),
            mapOf("Content-Type" to "application/json")
        )
        val json = JSONObject(body)
        return json.optJSONObject("task")?.toTask() ?: json.toTask()
    }

    suspend fun getTaskStatus(taskId: String): Task {
        val body = ApiClient.request("/tasks/$taskId")
        val json = JSONObject(body)
        return json.optJSONObject("task")?.toTask() ?: json.toTask()
    }

    suspend fun claimTask(taskId: String): Task {
        val body = ApiClient.request(
            "/tasks/$taskId/claim",
            "POST",
            "{}",
            mapOf("Content-Type" to "application/json")
        )
        val json = JSONObject(body)
        return json.optJSONObject("task")?.toTask() ?: json.toTask()
    }

    suspend fun confirmTask(taskId: String): Task {
        val body = ApiClient.request(
            "/tasks/$taskId/confirm",
            "POST",
            "{}",
            mapOf("Content-Type" to "application/json")
        )
        val json = JSONObject(body)
        return json.optJSONObject("task")?.toTask() ?: json.toTask()
    }

    suspend fun cancelTask(taskId: String) {
        ApiClient.request(
            "/tasks/$taskId/cancel",
            "POST",
            "{}",
            mapOf("Content-Type" to "application/json")
        )
    }

    private fun JSONObject.toDevice(): Device {
        val accountArray = optJSONArray("account_ids") ?: JSONArray()
        return Device(
            id = optString("id"),
            name = optString("name").ifEmpty { optString("display_username") },
            model = optString("model"),
            status = optString("status", "offline"),
            online = optBoolean("online") || optString("status").equals("online", true),
            accountIds = (0 until accountArray.length()).map { accountArray.optString(it) }
        )
    }

    private fun JSONObject.toAccount(): Account {
        return Account(
            id = optString("id"),
            displayName = optString("display_username")
                .ifEmpty { optString("nickname") }
                .ifEmpty { optString("account") },
            account = optString("account"),
            platform = optString("platform")
        )
    }

    private fun JSONObject.toTaskType(): TaskType {
        return TaskType(
            id = optString("id").ifEmpty { optString("type") },
            name = optString("name").ifEmpty { optString("task_type") },
            reward = optString("reward").ifEmpty { optString("reward_text") },
            needsManualConfirm = optBoolean("needs_manual_confirm")
                || optBoolean("manual_confirm")
                || optBoolean("needs_verify"),
            category = optString("category")
        )
    }

    private fun JSONObject.toTask(): Task {
        val accountArray = optJSONArray("account_ids") ?: JSONArray()
        val status = optString("status", "pending")
        return Task(
            id = optString("id"),
            type = optString("task_type").ifEmpty { optString("type") },
            typeName = optString("task_type_name")
                .ifEmpty { optString("type_name") }
                .ifEmpty { optString("task_type") },
            status = status,
            progress = optInt("progress", if (status == "success" || status == "completed") 100 else 0),
            reward = optString("reward").ifEmpty { optString("reward_text") },
            message = optString("message").ifEmpty { optString("log") },
            accountIds = (0 until accountArray.length()).map { accountArray.optString(it) },
            createdAt = optString("created_at")
        )
    }
}

package com.skyauto.app.ui

import androidx.lifecycle.ViewModel
import androidx.lifecycle.viewModelScope
import com.skyauto.app.data.Account
import com.skyauto.app.data.Device
import com.skyauto.app.data.Repository
import kotlinx.coroutines.flow.MutableStateFlow
import kotlinx.coroutines.flow.StateFlow
import kotlinx.coroutines.flow.asStateFlow
import kotlinx.coroutines.launch

class DeviceViewModel : ViewModel() {

    private val repo = Repository()

    private val _devices = MutableStateFlow<List<Device>>(emptyList())
    val devices: StateFlow<List<Device>> = _devices.asStateFlow()

    private val _accounts = MutableStateFlow<List<Account>>(emptyList())
    val accounts: StateFlow<List<Account>> = _accounts.asStateFlow()

    private val _loading = MutableStateFlow(false)
    val loading: StateFlow<Boolean> = _loading.asStateFlow()

    private val _error = MutableStateFlow<String?>(null)
    val error: StateFlow<String?> = _error.asStateFlow()

    fun load() {
        viewModelScope.launch {
            _loading.value = true
            _error.value = null
            try {
                repo.fetchSiteConfig()
                _devices.value = repo.getDevices()
                _accounts.value = repo.getAccounts()
            } catch (e: Exception) {
                _error.value = e.message ?: "加载失败"
            } finally {
                _loading.value = false
            }
        }
    }

    fun createDevice(name: String, model: String) {
        viewModelScope.launch {
            _loading.value = true
            _error.value = null
            try {
                val created = repo.createDevice(name, model)
                _devices.value = listOf(created) + _devices.value.filter { it.id != created.id }
            } catch (e: Exception) {
                _error.value = e.message ?: "创建设备失败"
            } finally {
                _loading.value = false
            }
        }
    }

    fun deleteDevice(deviceId: String) {
        viewModelScope.launch {
            try {
                repo.deleteDevice(deviceId)
                _devices.value = _devices.value.filter { it.id != deviceId }
            } catch (e: Exception) {
                _error.value = e.message ?: "删除失败"
            }
        }
    }

    fun setDeviceOnline(deviceId: String, online: Boolean) {
        viewModelScope.launch {
            try {
                repo.setDeviceOnline(deviceId, online)
                _devices.value = _devices.value.map {
                    if (it.id == deviceId) it.copy(online = online, status = if (online) "online" else "offline") else it
                }
            } catch (e: Exception) {
                _error.value = e.message ?: "模拟注入失败"
            }
        }
    }

    fun bindAccounts(deviceId: String, accountIds: List<String>) {
        viewModelScope.launch {
            _loading.value = true
            _error.value = null
            try {
                val current = _devices.value.firstOrNull { it.id == deviceId } ?: return@launch
                val target = accountIds.toSet()
                val existing = current.accountIds.toSet()
                (target - existing).forEach { repo.bindAccount(deviceId, it) }
                (existing - target).forEach { repo.unbindAccount(deviceId, it) }
                _devices.value = repo.getDevices()
            } catch (e: Exception) {
                _error.value = e.message ?: "绑定失败"
            } finally {
                _loading.value = false
            }
        }
    }
}

package com.skyauto.app.ui

import androidx.lifecycle.ViewModel
import androidx.lifecycle.viewModelScope
import com.skyauto.app.data.PollState
import com.skyauto.app.data.Repository
import com.skyauto.app.data.Task
import com.skyauto.app.data.TaskType
import kotlinx.coroutines.Job
import kotlinx.coroutines.delay
import kotlinx.coroutines.flow.MutableStateFlow
import kotlinx.coroutines.flow.StateFlow
import kotlinx.coroutines.flow.asStateFlow
import kotlinx.coroutines.isActive
import kotlinx.coroutines.launch

class TaskViewModel : ViewModel() {

    private val repo = Repository()

    private val _taskTypes = MutableStateFlow<List<TaskType>>(emptyList())
    val taskTypes: StateFlow<List<TaskType>> = _taskTypes.asStateFlow()

    private val _tasks = MutableStateFlow<List<Task>>(emptyList())
    val tasks: StateFlow<List<Task>> = _tasks.asStateFlow()

    private val _pollState = MutableStateFlow(PollState.IDLE)
    val pollState: StateFlow<PollState> = _pollState.asStateFlow()

    private val _confirmTask = MutableStateFlow<Task?>(null)
    val confirmTask: StateFlow<Task?> = _confirmTask.asStateFlow()

    private val _loading = MutableStateFlow(false)
    val loading: StateFlow<Boolean> = _loading.asStateFlow()

    private val _error = MutableStateFlow<String?>(null)
    val error: StateFlow<String?> = _error.asStateFlow()

    private var pollJob: Job? = null

    fun load() {
        viewModelScope.launch {
            _loading.value = true
            _error.value = null
            try {
                repo.fetchSiteConfig()
                _taskTypes.value = repo.getTaskTypes()
                _tasks.value = repo.getTasks()
                _confirmTask.value = _tasks.value.firstOrNull { it.needsManualConfirm }
            } catch (e: Exception) {
                _error.value = e.message ?: "加载失败"
            } finally {
                _loading.value = false
            }
        }
    }

    fun createTask(type: String, accountIds: List<String>, deviceId: String?) {
        viewModelScope.launch {
            _loading.value = true
            _error.value = null
            try {
                val task = repo.createTask(type, accountIds, deviceId)
                _tasks.value = listOf(task) + _tasks.value.filter { it.id != task.id }
                _confirmTask.value = null
                _pollState.value = PollState.POLLING
                startPolling(task.id)
            } catch (e: Exception) {
                _error.value = e.message ?: "创建任务失败"
                _pollState.value = PollState.ERROR
            } finally {
                _loading.value = false
            }
        }
    }

    fun startPolling(taskId: String) {
        pollJob?.cancel()
        _pollState.value = PollState.POLLING
        pollJob = viewModelScope.launch {
            while (isActive) {
                try {
                    val task = repo.getTaskStatus(taskId)
                    _tasks.value = _tasks.value.map { if (it.id == task.id) task else it }

                    if (task.needsManualConfirm) {
                        _pollState.value = PollState.WAITING_MANUAL
                        _confirmTask.value = task
                        break
                    }

                    if (task.isTerminal) {
                        _pollState.value = PollState.DONE
                        break
                    }

                    delay(2000)
                } catch (e: Exception) {
                    _error.value = e.message ?: "轮询失败"
                    _pollState.value = PollState.ERROR
                    break
                }
            }
        }
    }

    fun refreshTasks() {
        viewModelScope.launch {
            try {
                _tasks.value = repo.getTasks()
                _confirmTask.value = _tasks.value.firstOrNull { it.needsManualConfirm }
            } catch (e: Exception) {
                _error.value = e.message ?: "刷新失败"
            }
        }
    }

    fun claim(task: Task) {
        viewModelScope.launch {
            try {
                val updated = repo.claimTask(task.id)
                _tasks.value = _tasks.value.map { if (it.id == updated.id) updated else it }
            } catch (e: Exception) {
                _error.value = e.message ?: "领取失败"
            }
        }
    }

    fun cancel(task: Task) {
        viewModelScope.launch {
            try {
                repo.cancelTask(task.id)
                _tasks.value = _tasks.value.filter { it.id != task.id }
                if (_confirmTask.value?.id == task.id) {
                    _confirmTask.value = null
                    _pollState.value = PollState.DONE
                }
            } catch (e: Exception) {
                _error.value = e.message ?: "取消失败"
            }
        }
    }

    fun confirmManual() {
        val task = _confirmTask.value ?: return
        viewModelScope.launch {
            _loading.value = true
            _error.value = null
            try {
                val updated = repo.confirmTask(task.id)
                _tasks.value = _tasks.value.map { if (it.id == updated.id) updated else it }
                _confirmTask.value = null
                _pollState.value = PollState.POLLING
                startPolling(updated.id)
            } catch (e: Exception) {
                _error.value = e.message ?: "确认失败"
                _pollState.value = PollState.ERROR
            } finally {
                _loading.value = false
            }
        }
    }

    fun dismissConfirm() {
        _confirmTask.value = null
    }
}
