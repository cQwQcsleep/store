package com.skyangin.client.data

import android.util.Base64
import kotlinx.coroutines.CancellationException
import kotlinx.coroutines.CoroutineScope
import kotlinx.coroutines.Deferred
import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.SupervisorJob
import kotlinx.coroutines.async
import kotlinx.coroutines.withContext
import kotlinx.serialization.decodeFromString
import kotlinx.serialization.json.Json
import okhttp3.OkHttpClient
import okhttp3.Request
import java.security.SecureRandom
import javax.crypto.Cipher
import javax.crypto.SecretKey
import javax.crypto.spec.GCMParameterSpec
import javax.crypto.spec.SecretKeySpec

/**
 * 复刻前端第8393号模块的 AES-256-GCM 语义：
 * 随机12字节IV，输出 base64(IV || ciphertext)。
 */
class CryptoManager(
    private val client: OkHttpClient,
    private val json: Json
) {
    @Volatile
    private var key: SecretKey? = null

    @Volatile
    private var keyFetching: Deferred<SecretKey?>? = null

    private val scope = CoroutineScope(SupervisorJob() + Dispatchers.IO)

    fun clear() {
        key = null
        keyFetching?.cancel()
        keyFetching = null
    }

    private suspend fun getKey(apiBase: String): SecretKey? {
        key?.let { return it }

        keyFetching?.let { deferred ->
            return try {
                deferred.await()
            } catch (e: CancellationException) {
                null
            }
        }

        val deferred = scope.async {
            try {
                val url = "$apiBase/crypto/key"
                val request = Request.Builder().url(url).get().build()
                client.newCall(request).execute().use { response ->
                    if (!response.isSuccessful) return@async null
                    val body = response.body?.string() ?: return@async null
                    val keyResponse = json.decodeFromString<KeyResponse>(body)
                    if (!keyResponse.success || keyResponse.key.isNullOrEmpty()) return@async null
                    val rawKey = Base64.decode(keyResponse.key, Base64.NO_WRAP)
                    SecretKeySpec(rawKey, "AES")
                }
            } catch (e: Exception) {
                null
            }
        }

        keyFetching = deferred
        return try {
            deferred.await()?.also { key = it }
        } finally {
            keyFetching = null
        }
    }

    suspend fun encrypt(plainText: String, apiBase: String): String? =
        withContext(Dispatchers.Default) {
            val secret = getKey(apiBase) ?: return@withContext null
            try {
                val iv = ByteArray(12).also { SecureRandom().nextBytes(it) }
                val cipher = Cipher.getInstance("AES/GCM/NoPadding")
                cipher.init(Cipher.ENCRYPT_MODE, secret, GCMParameterSpec(128, iv))
                val ciphertext = cipher.doFinal(plainText.toByteArray(Charsets.UTF_8))
                val combined = ByteArray(12 + ciphertext.size)
                System.arraycopy(iv, 0, combined, 0, 12)
                System.arraycopy(ciphertext, 0, combined, 12, ciphertext.size)
                Base64.encodeToString(combined, Base64.NO_WRAP)
            } catch (e: Exception) {
                null
            }
        }

    suspend fun decrypt(encrypted: String, apiBase: String): String? =
        withContext(Dispatchers.Default) {
            val secret = getKey(apiBase) ?: return@withContext null
            try {
                val data = Base64.decode(encrypted, Base64.NO_WRAP)
                if (data.size <= 12) return@withContext null
                val iv = data.copyOfRange(0, 12)
                val ciphertext = data.copyOfRange(12, data.size)
                val cipher = Cipher.getInstance("AES/GCM/NoPadding")
                cipher.init(Cipher.DECRYPT_MODE, secret, GCMParameterSpec(128, iv))
                String(cipher.doFinal(ciphertext), Charsets.UTF_8)
            } catch (e: Exception) {
                null
            }
        }
}

package com.skyangin.client.data

import android.content.Context
import android.content.SharedPreferences
import androidx.security.crypto.EncryptedSharedPreferences
import androidx.security.crypto.MasterKey
import kotlinx.serialization.Serializable
import kotlinx.serialization.decodeFromString
import kotlinx.serialization.encodeToString
import kotlinx.serialization.json.Json
import okhttp3.Cookie
import okhttp3.CookieJar
import okhttp3.HttpUrl

class PersistentCookieJar(context: Context) : CookieJar {

    private val prefs: SharedPreferences
    private val json = Json { ignoreUnknownKeys = true }
    private val cache: MutableList<StoredCookie>

    init {
        val masterKey = MasterKey.Builder(context)
            .setKeyScheme(MasterKey.KeyScheme.AES256_GCM)
            .build()
        prefs = EncryptedSharedPreferences.create(
            context,
            "sky_cookies",
            masterKey,
            EncryptedSharedPreferences.PrefKeyEncryptionScheme.AES256_SIV,
            EncryptedSharedPreferences.PrefValueEncryptionScheme.AES256_GCM
        )
        cache = load()
    }

    @Synchronized
    override fun saveFromResponse(url: HttpUrl, cookies: List<Cookie>) {
        cookies.forEach { cookie ->
            cache.removeAll { it.name == cookie.name && it.domain == cookie.domain }
            cache.add(cookie.toStored())
        }
        persist()
    }

    @Synchronized
    override fun loadForRequest(url: HttpUrl): List<Cookie> {
        val now = System.currentTimeMillis()
        return cache
            .filter { it.expiresAt > now && it.toCookie().matches(url) }
            .map { it.toCookie() }
    }

    @Synchronized
    fun clear() {
        cache.clear()
        prefs.edit().clear().apply()
    }

    private fun persist() {
        val raw = json.encodeToString(StoredCookieList(cache.toList()))
        prefs.edit().putString(KEY, raw).apply()
    }

    private fun load(): MutableList<StoredCookie> {
        val raw = prefs.getString(KEY, null) ?: return mutableListOf()
        return try {
            json.decodeFromString<StoredCookieList>(raw).cookies.toMutableList()
        } catch (e: Exception) {
            mutableListOf()
        }
    }

    private fun Cookie.toStored() = StoredCookie(
        name = name,
        value = value,
        domain = domain,
        path = path,
        expiresAt = expiresAt,
        secure = secure,
        httpOnly = httpOnly,
        hostOnly = hostOnly
    )

    private fun StoredCookie.toCookie(): Cookie {
        val builder = Cookie.Builder()
            .name(name)
            .value(value)
            .path(path)
            .expiresAt(expiresAt)
            .secure(secure)
            .httpOnly(httpOnly)
        return if (hostOnly) {
            builder.hostOnlyDomain(domain).build()
        } else {
            builder.domain(domain).build()
        }
    }

    @Serializable
    private data class StoredCookie(
        val name: String,
        val value: String,
        val domain: String,
        val path: String,
        val expiresAt: Long,
        val secure: Boolean,
        val httpOnly: Boolean,
        val hostOnly: Boolean
    )

    @Serializable
    private data class StoredCookieList(val cookies: List<StoredCookie>)

    private companion object {
        private const val KEY = "cookies"
    }
}

package com.skyangin.client.data

import kotlinx.coroutines.CoroutineScope
import kotlinx.coroutines.Deferred
import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.async
import kotlinx.coroutines.withContext
import kotlinx.serialization.decodeFromString
import kotlinx.serialization.json.Json
import kotlinx.serialization.json.buildJsonObject
import kotlinx.serialization.json.put
import okhttp3.MediaType.Companion.toMediaType
import okhttp3.OkHttpClient
import okhttp3.Request
import okhttp3.RequestBody.Companion.toRequestBody
import java.util.concurrent.ConcurrentHashMap

class ApiClient(
    baseUrl: String,
    private val client: OkHttpClient,
    private val json: Json,
    private val cookieJar: PersistentCookieJar
) {
    private val apiBase = baseUrl.trimEnd('/')
    private val apiPrefix = "$apiBase/api"
    private val jsonMediaType = "application/json; charset=utf-8".toMediaType()
    private val authHeaders = mapOf("Content-Type" to "application/json")
    private val crypto = CryptoManager(client, json)
    private val cache = ConcurrentHashMap<String, CachedResponse>()
    private val inFlight = ConcurrentHashMap<String, Deferred<String>>()

    suspend fun login(account: String, loginType: String, password: String, deviceId: String): AuthResponse {
        val body = buildJsonObject {
            put("account", account)
            put("login_type", loginType)
            put("password", password)
            put("device_id", deviceId)
        }.toString()

        val response = execute(
            path = "/api/auth/login",
            method = "POST",
            body = body,
            headers = authHeaders,
            authSensitive = true
        )
        return json.decodeFromString<AuthResponse>(response)
    }

    suspend fun register(
        account: String,
        loginType: String,
        password: String,
        deviceId: String,
        code: String? = null
    ): AuthResponse {
        val body = buildJsonObject {
            put("account", account)
            put("login_type", loginType)
            put("password", password)
            put("device_id", deviceId)
            if (!code.isNullOrBlank()) put("code", code)
        }.toString()

        val response = execute(
            path = "/api/auth/register",
            method = "POST",
            body = body,
            headers = authHeaders,
            authSensitive = true
        )
        return json.decodeFromString<AuthResponse>(response)
    }

    suspend fun sendResetCode(email: String): MessageResponse {
        val response = execute(
            path = "/api/auth/send-reset-code/$email",
            method = "POST",
            body = "{}",
            headers = authHeaders
        )
        return json.decodeFromString<MessageResponse>(response)
    }

    suspend fun logout() {
        try {
            execute(
                path = "/api/auth/logout",
                method = "POST",
                body = null,
                authSensitive = true
            )
        } catch (_: Exception) {
        }
    }

    suspend fun me(): User? {
        val response = execute(
            path = "/api/auth/me",
            method = "GET",
            body = null
        )
        return json.decodeFromString<MeResponse>(response).effectiveUser
    }

    fun clearCookies() {
        cookieJar.clear()
    }

    private suspend fun execute(
        path: String,
        method: String,
        body: String?,
        headers: Map<String, String> = emptyMap(),
        authSensitive: Boolean = false
    ): String {
        val url = if (path.startsWith("http")) path else apiBase + path

        if (authSensitive) crypto.clear()

        return try {
            if (method.equals("GET", true) && body == null && path.contains("/config/site")) {
                getCached(url, 10_000L) {
                    fetchInternal(url, method, body, headers, authSensitive)
                }
            } else {
                fetchInternal(url, method, body, headers, authSensitive)
            }
        } finally {
            if (authSensitive) crypto.clear()
        }
    }

    private suspend fun fetchInternal(
        url: String,
        method: String,
        body: String?,
        headers: Map<String, String>,
        authSensitive: Boolean
    ): String = withContext(Dispatchers.IO) {
        val shouldEncrypt = shouldEncrypt(url, body)
        val builder = Request.Builder().url(url)
        headers.forEach { (k, v) -> builder.header(k, v) }

        var finalBody = body
        if (shouldEncrypt && body != null) {
            val encrypted = crypto.encrypt(body, apiPrefix)
            if (encrypted != null) {
                builder.header("X-Encrypted", "1")
                finalBody = encrypted
            }
        }

        val requestBody = when {
            finalBody != null -> finalBody.toRequestBody(jsonMediaType)
            method.equals("POST", true) ||
                method.equals("PUT", true) ||
                method.equals("PATCH", true) -> ByteArray(0).toRequestBody(null)
            else -> null
        }
        builder.method(method, requestBody)

        client.newCall(builder.build()).execute().use { response ->
            val responseBody = response.body?.string() ?: ""

            // 与前端模块一致：auth 敏感请求在拿到响应后、解密前清空 key。
            if (authSensitive) crypto.clear()

            if (shouldEncrypt && response.header("X-Encrypted") == "1") {
                val decrypted = crypto.decrypt(responseBody, apiPrefix)
                if (decrypted != null) return@withContext decrypted
            }
            return@withContext responseBody
        }
    }

    private fun shouldEncrypt(url: String, body: String?): Boolean {
        if (!url.contains("/api/")) return false
        if (url.contains("/api/crypto/key")) return false
        return body != null
    }

    private suspend fun getCached(
        url: String,
        millis: Long,
        fetcher: suspend () -> String
    ): String {
        cache[url]?.let {
            if (System.currentTimeMillis() < it.expiresAt) return it.body
        }

        inFlight[url]?.let { return it.await() }

        val deferred = CoroutineScope(Dispatchers.IO).async { fetcher() }
        inFlight[url] = deferred

        return try {
            val body = deferred.await()
            cache[url] = CachedResponse(body, System.currentTimeMillis() + millis)
            body
        } finally {
            inFlight.remove(url)
        }
    }

    private data class CachedResponse(val body: String, val expiresAt: Long)
}

package com.skyangin.client.data

import android.content.Context
import androidx.datastore.core.DataStore
import androidx.datastore.preferences.core.Preferences
import androidx.datastore.preferences.core.edit
import androidx.datastore.preferences.core.stringPreferencesKey
import androidx.datastore.preferences.preferencesDataStore
import kotlinx.coroutines.flow.MutableStateFlow
import kotlinx.coroutines.flow.StateFlow
import kotlinx.coroutines.flow.asStateFlow
import kotlinx.coroutines.flow.first
import kotlinx.serialization.decodeFromString
import kotlinx.serialization.encodeToString
import kotlinx.serialization.json.Json
import java.util.UUID

private val Context.dataStore: DataStore<Preferences> by preferencesDataStore(name = "session")

class SessionManager(
    private val context: Context,
    private val apiClient: ApiClient
) {
    private val json = Json {
        ignoreUnknownKeys = true
        encodeDefaults = false
    }

    private val _state = MutableStateFlow<SessionState>(SessionState.Loading)
    val state: StateFlow<SessionState> = _state.asStateFlow()

    suspend fun restore() {
        val prefs = context.dataStore.data.first()
        val userJson = prefs[USER_JSON] ?: run {
            _state.value = SessionState.Unauthenticated
            return
        }

        _state.value = SessionState.Loading

        val cachedUser = try {
            json.decodeFromString<User>(userJson)
        } catch (e: Exception) {
            null
        }

        if (cachedUser == null) {
            clear()
            return
        }

        val freshUser = try {
            apiClient.me()
        } catch (e: Exception) {
            null
        }

        if (freshUser != null) {
            saveUser(freshUser)
            _state.value = SessionState.Authenticated(freshUser)
        } else {
            clear()
        }
    }

    suspend fun onAuthenticated(user: User) {
        saveUser(user)
        _state.value = SessionState.Authenticated(user)
    }

    suspend fun logout() {
        try {
            apiClient.logout()
        } catch (_: Exception) {
        }
        clear()
    }

    suspend fun getDeviceId(): String {
        val prefs = context.dataStore.data.first()
        prefs[DEVICE_ID]?.let { return it }

        val newId = UUID.randomUUID().toString()
        context.dataStore.edit { it[DEVICE_ID] = newId }
        return newId
    }

    private suspend fun saveUser(user: User) {
        context.dataStore.edit { it[USER_JSON] = json.encodeToString(user) }
    }

    suspend fun clear() {
        context.dataStore.edit { it.remove(USER_JSON) }
        _state.value = SessionState.Unauthenticated
        apiClient.clearCookies()
    }

    private companion object {
        val USER_JSON = stringPreferencesKey("user_json")
        val DEVICE_ID = stringPreferencesKey("device_id")

) {
    val state by sessionManager.state.collectAsState()
    val scope = rememberCoroutineScope()

    LaunchedEffect(Unit) {
        sessionManager.restore()
    }

    when (val s = state) {
        is SessionState.Loading -> SplashScreen()

        is SessionState.Authenticated -> WelcomeScreen(
            user = s.user,
            onLogout = {
                scope.launch { sessionManager.logout() }
            }
        )

        is SessionState.Unauthenticated -> AuthNavHost(
            sessionManager = sessionManager,
            apiClient = apiClient
        )
    }
}

@Composable
private fun AuthNavHost(
    sessionManager: SessionManager,
    apiClient: ApiClient
) {
    val navController = rememberNavController()
    val scope = rememberCoroutineScope()

    NavHost(
        navController = navController,
        startDestination = "login"
    ) {
        composable("login") {
            LoginScreen(
                apiClient = apiClient,
                sessionManager = sessionManager,
                onLoginSuccess = { user ->
                    scope.launch {
                        sessionManager.onAuthenticated(user)
                    }
                },
                onNavigateRegister = { navController.navigate("register") },
                onNavigateForgot = { navController.navigate("forgot") }
            )
        }

        composable("register") {
            RegisterScreen(
                apiClient = apiClient,
                sessionManager = sessionManager,
                onRegistered = { user ->
                    scope.launch {
                        sessionManager.onAuthenticated(user)
                    }
                },
                onBack = { navController.popBackStack() }
            )
        }

        composable("forgot") {
            ForgotPasswordScreen(
                apiClient = apiClient,
                onBack = { navController.popBackStack() }
            )
        }
    }
}

@Composable
private fun SplashScreen() {
    Box(
        modifier = Modifier.fillMaxSize(),
        contentAlignment = Alignment.Center
    ) {
        CircularProgressIndicator()
    }
}
